rm -f ImplicitTaintPropagation.so
rm -f dfsan.ll
rm -f instrumented.ll
rm -f implicit_test

clang++ \
  -std=c++17 \
  -fPIC \
  -shared \
  ImplicitTaintPropagation.cpp \
  -o ImplicitTaintPropagation.so \
  $(llvm-config \
    --cxxflags \
    --ldflags \
    --system-libs \
    --libs core analysis passes)

clang \
  -O0 \
  -g \
  -Xclang -disable-O0-optnone \
  -fsanitize=dataflow \
  -mllvm -dfsan-conditional-callbacks \
  -S \
  -emit-llvm \
  test.c \
  -o dfsan.ll

opt \
  -load-pass-plugin=./ImplicitTaintPropagation.so \
  -passes=implicit-taint \
  -S \
  dfsan.ll \
  -o instrumented.ll

DFSAN_RT="$(clang -print-resource-dir)/lib/linux/libclang_rt.dfsan-x86_64.a"

clang \
  -fno-sanitize=dataflow \
  instrumented.ll \
  "$DFSAN_RT" \
  -pthread \
  -lm \
  -ldl \
  -lrt \
  -lresolv \
  -o implicit_test

./implicit_test
